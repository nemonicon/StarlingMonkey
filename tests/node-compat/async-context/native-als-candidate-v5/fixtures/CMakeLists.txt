cmake_minimum_required(VERSION 3.5)
project(ALSStrictInitializationGuardPolicy LANGUAGES NONE)

option(TEST_NODE_TARGET "Create the Node compatibility target" ON)
option(TEST_EXTENSION_TARGET "Create the genuine policy producer target" ON)
option(TEST_TICKS "Enable the private tick prerequisite" ON)
set(TEST_MARKER "emit" CACHE STRING "emit, absent, disabled or multiple")
set(TEST_CMAKE_VARIABLE "unset" CACHE STRING "unset, ON or OFF")
if(TEST_NODE_TARGET)
    add_library(builtin_node_compat INTERFACE)
endif()
if(TEST_EXTENSION_TARGET)
    add_library(extension_api INTERFACE)
    if(TEST_MARKER STREQUAL "emit" OR TEST_MARKER STREQUAL "multiple")
        target_compile_definitions(extension_api INTERFACE NODE_COMPAT_STRICT_INITIALIZATION=1)
    elseif(TEST_MARKER STREQUAL "disabled")
        target_compile_definitions(extension_api INTERFACE NODE_COMPAT_STRICT_INITIALIZATION=0)
    endif()
    if(TEST_MARKER STREQUAL "multiple")
        target_compile_definitions(extension_api INTERFACE NODE_COMPAT_INTERNAL_TICKS=1)
    endif()
endif()
set(NODE_COMPAT_INTERNAL_TICKS "${TEST_TICKS}")
if(TEST_CMAKE_VARIABLE STREQUAL "unset")
    unset(NODE_COMPAT_STRICT_INITIALIZATION)
else()
    set(NODE_COMPAT_STRICT_INITIALIZATION "${TEST_CMAKE_VARIABLE}")
endif()
# This file is the exact extracted active ALS guard, not a model in another
# language and not the original native project's full configuration graph.
include("${TEST_GUARD_PATH}")
get_target_property(OBSERVED_DEFINITIONS extension_api INTERFACE_COMPILE_DEFINITIONS)
file(WRITE "${CMAKE_BINARY_DIR}/observed-definitions.txt" "${OBSERVED_DEFINITIONS}\n")
