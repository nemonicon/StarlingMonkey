def row(path):
    p=Path(path);h=hashlib.sha256()
    with p.open('rb')as f:
        for b in iter(lambda:f.read(4*1024**2),b''):h.update(b)
    return dict(path=str(p),sha256=h.hexdigest(),bytes=p.stat().st_size)
def checked(r):
    p=Path(r['path']);assert p.is_absolute()and'..'not in p.parts
    assert all(not x.is_symlink()for x in(p,*p.parents));s=p.stat()
    assert row(p)=={k:r[k]for k in('path','sha256','bytes')};t=p.stat()
    assert(s.st_dev,s.st_ino,s.st_mode,s.st_nlink,s.st_size,s.st_mtime_ns,s.st_ctime_ns)==(t.st_dev,t.st_ino,t.st_mode,t.st_nlink,t.st_size,t.st_mtime_ns,t.st_ctime_ns)
    return p
def write(p,value):
    p=Path(p);data=value if isinstance(value,bytes)else(json.dumps(value,indent=2)+'\n').encode()
    fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
    with os.fdopen(fd,'wb')as f:os.fchmod(f.fileno(),0o600);f.write(data);f.flush();os.fsync(f.fileno())
