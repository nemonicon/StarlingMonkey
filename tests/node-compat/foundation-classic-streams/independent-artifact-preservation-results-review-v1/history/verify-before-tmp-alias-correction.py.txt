def need(value, message):
    if not value:
        raise AssertionError(message)


def digest(value):
    return hashlib.sha256(value).hexdigest()


def identity(info):
    return (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns, info.st_ctime_ns,
            stat.S_IMODE(info.st_mode), info.st_nlink)


def safe(name):
    path = PurePosixPath(name)
    need(type(name) is str and name and '\\' not in name and '\0' not in name and
         not path.is_absolute() and str(path) == name and
         all(p not in ('', '.', '..') for p in path.parts), 'Safe archive relative path')
    return path.parts


def absolute_fd(path):
    path = Path(path)
    need(path.is_absolute() and str(path) == os.path.normpath(path), 'Canonical absolute witness')
    directory = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        for part in path.parts[1:-1]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory)
            os.close(directory)
            directory = child
        return os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=directory)
    finally:
        os.close(directory)


def stream(fd, expected, size, collect=False):
    before = os.fstat(fd)
    need(stat.S_ISREG(before.st_mode) and before.st_size == size, 'Regular expected-size file')
    h, parts = hashlib.sha256(), []
    while True:
        value = os.read(fd, 1024 * 1024)
        if not value:
            break
        h.update(value)
        if collect:
            parts.append(value)
    need(identity(before) == identity(os.fstat(fd)), 'File changed while hashing')
    need(h.hexdigest() == expected, 'SHA mismatch')
    return (b''.join(parts) if collect else None), before


def external(path, expected=None, size=None, collect=True):
    path = Path(path)
    fd = absolute_fd(path)
    try:
        before = os.fstat(fd)
        need(stat.S_ISREG(before.st_mode), 'External regular file')
        if expected is None:
            expected = hashlib.sha256(path.read_bytes()).hexdigest()
        if size is None:
            size = before.st_size
        value, info = stream(fd, expected, size, collect)
        witnesses[str(path)] = {'path': str(path), 'sha256': expected, 'bytes': size}
        return value
    finally:
        os.close(fd)

def closure(fd):
    files, directories = {}, {}
    def walk(directory, prefix):
        for name in sorted(os.listdir(directory)):
            info = os.stat(name, dir_fd=directory, follow_symlinks=False)
            relative = prefix + name
            safe(relative)
            need(not stat.S_ISLNK(info.st_mode), 'No archive symlink')
            if stat.S_ISDIR(info.st_mode):
                need(stat.S_IMODE(info.st_mode) == 0o700, 'Private subdirectory mode')
                child = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory)
                try:
                    actual = os.fstat(child)
                    need((actual.st_dev, actual.st_ino) == (info.st_dev, info.st_ino), 'Directory race')
                    directories[relative] = (info.st_dev, info.st_ino, stat.S_IMODE(info.st_mode))
                    walk(child, relative + '/')
                finally:
                    os.close(child)
            else:
                need(stat.S_ISREG(info.st_mode) and info.st_nlink == 1 and
                     stat.S_IMODE(info.st_mode) == 0o600, 'Private regular unshared file')
                files[relative] = identity(info)
    walk(fd, '')
    return files, directories

def archived(name, expected, size, collect=True):
    parts = safe(name)
    directory, fd = os.dup(root_fd), None
    try:
        for part in parts[:-1]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory)
            os.close(directory)
            directory = child
        fd = os.open(parts[-1], os.O_RDONLY | os.O_NOFOLLOW, dir_fd=directory)
        value, info = stream(fd, expected, size, collect)
        need(info.st_nlink == 1 and stat.S_IMODE(info.st_mode) == 0o600, 'Owned copied file metadata')
        root_identity()
        return value
    finally:
        if fd is not None:
            os.close(fd)
        os.close(directory)
