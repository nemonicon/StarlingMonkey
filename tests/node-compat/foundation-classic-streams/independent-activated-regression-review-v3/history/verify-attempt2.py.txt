def digest(data):
    return hashlib.sha256(data).hexdigest()


def absolute(path):
    return Path(os.path.abspath(path))


def regular(path, expected=None, size=None):
    path = absolute(path)
    before = path.lstat()
    assert stat.S_ISREG(before.st_mode), str(path)
    if size is not None:
        assert before.st_size == size, str(path)
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(chunk)
    after = path.lstat()
    assert (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns) == (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns), str(path)
    row = {'path': str(path), 'sha256': h.hexdigest(), 'bytes': before.st_size}
    assert expected is None or expected == row['sha256'], str(path)
    if str(path) in checks:
        assert checks[str(path)] == row
    checks[str(path)] = row
    return row


def document(path, expected=None, size=None):
    row = regular(path, expected, size)
    data = Path(path).read_bytes()
    assert digest(data) == row['sha256']
    return json.loads(data)


def below(root, path):
    root, path = absolute(root), absolute(path)
    assert path != root and path.is_relative_to(root), (str(root), str(path))
    assert root.is_dir() and not root.is_symlink(), str(root)
    cursor = path
    while cursor != root:
        assert not cursor.is_symlink(), str(cursor)
        cursor = cursor.parent


def membership(root):
    assert root.is_dir() and not root.is_symlink(), str(root)
    rows = []
    for directory, dirs, files in os.walk(root, followlinks=False):
        for name in sorted(dirs + files):
            path = Path(directory) / name
            mode = path.lstat().st_mode
            kind = ('symlink' if stat.S_ISLNK(mode) else 'directory' if stat.S_ISDIR(mode) else 'file' if stat.S_ISREG(mode) else None)
            assert kind, str(path)
            rows.append({'path': str(path.relative_to(root)), 'kind': kind})
        dirs[:] = [name for name in dirs if not (Path(directory) / name).is_symlink()]
    return sorted(rows, key=lambda row: row['path'])



def indexed(root, expected, count=None, exact_membership=False):
    value = document(root/'index.json', expected)
    rows = {r['path']:r for r in value['files']}
    assert len(rows) == len(value['files'])
    assert count is None or count == len(rows)
    for name, row in rows.items():
        assert not Path(name).is_absolute() and '..' not in Path(name).parts and '\\' not in name
        path = root/name
        below(root,path)
        regular(path,row['sha256'],row['bytes'])
    if exact_membership:
        observed=membership(root)
        assert {r['path'] for r in observed if r['kind']=='file'} == set(rows)|{'index.json'}
        assert not any(r['kind']=='symlink' for r in observed)
    return value, rows


def seal(path, expected, count):
    value=document(path,expected)
    rows={}
    for row in value['inputs']:
        target=absolute(path.parent/row['path'])
        assert target not in rows
        rows[target]=row
        regular(target,row['sha256'],row['bytes'])
    assert len(rows)==count
    return value,rows


def transform(before, edits):
    after=before;applications=[]
    for edit in edits:
        assert edit['old'] and after.count(edit['old'])==edit.get('occurrences',1)
        positions=[m.start() for m in re.finditer(re.escape(edit['old']),after)]
        delta=len(edit['new'])-len(edit['old'])
        applications.append((edit,[p+i*delta for i,p in enumerate(positions)],after))
        after=after.replace(edit['old'],edit['new'])
    reverse=after
    for edit,positions,previous in reversed(applications):
        for position in reversed(positions):
            assert reverse[position:position+len(edit['new'])]==edit['new']
            reverse=reverse[:position]+edit['old']+reverse[position+len(edit['new']):]
        assert reverse==previous
    assert reverse==before
    return after
