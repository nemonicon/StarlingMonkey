def need(value, message):
    if not value:
        raise AssertionError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def read(path, expected=None, size=None):
    path = Path(path)
    before = path.lstat()
    need(stat.S_ISREG(before.st_mode), 'Regular byte witness: ' + str(path))
    data = path.read_bytes()
    after = path.lstat()
    need((before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns, before.st_ctime_ns) ==
         (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns, after.st_ctime_ns), 'Changed read')
    digest = sha(data)
    need(expected is None or digest == expected, 'Hash: ' + str(path))
    need(size is None or len(data) == size, 'Size: ' + str(path))
    witnesses[str(path)] = {'path': str(path), 'sha256': digest, 'bytes': len(data)}
    return data


def load(path, expected=None, size=None):
    return json.loads(read(path, expected, size))


def safe_name(name):
    p = PurePosixPath(name)
    need(type(name) is str and name and '\\' not in name and '\0' not in name and
         not p.is_absolute() and str(p) == name and all(x not in ('', '.', '..') for x in p.parts),
         'Finite relative name')


def packet(root, expected):
    index = load(root / 'index.json', expected)
    names = set()
    for row in index['files']:
        safe_name(row['path'])
        need(row['path'] not in names, 'Unique index member')
        names.add(row['path'])
        read(root / row['path'], row['sha256'], row['bytes'])
    return index


def passed(name, **details):
    checks.append({'check': name, 'passed': True, **details})
